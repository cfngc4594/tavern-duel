func vec(d: Dictionary)->Vector3:return Vector3(d.x,d.y,d.z)
func trs(chain: Array)->Transform3D:
 var t:=Transform3D.IDENTITY
 for d in chain:
  var q: Dictionary=d.m_LocalRotation
  t=t*Transform3D(Basis(Quaternion(q.x,q.y,q.z,q.w)).scaled(vec(d.m_LocalScale)),vec(d.m_LocalPosition))
 return t
func vertices(d: Dictionary)->PackedVector3Array:
 var v:=PackedVector3Array()
 for a in d.vertices:v.append(Vector3(a[0],a[1],a[2]))
 return v
func indices(tris: Array)->PackedInt32Array:
 var ix:=PackedInt32Array()
 for t in tris:
  for i in t:ix.append(int(i))
 return ix
