func _attachments(subject: String,actor: Dictionary) -> Array:
 var result: Array=[]
 if not is_instance_valid(game.native_status):return result
 for effect in game.native_status.effects.effects:
  if effect.get("subject","")==subject and not effect.get("cached_none",false) and not effect.state.is_empty():
   var attachment: Dictionary=effect.state.duplicate()
   # Minion status projection is assigned the exact Actor transform each frame.
   # Hero status uses the existing registration before the original hero root's
   # authored local scale1.11; cancel only that constant local root transform.
   attachment.actor_relative_anchor=effect.projection
   attachment.actor_relative_transform=actor.root.transform.affine_inverse() if subject.begins_with("hero:") else Transform3D.IDENTITY
   result.append(attachment)
 return result

func _retire(id):
 if renderer.entries.has(id):
  var mesh=renderer.entries[id].mesh
  if is_instance_valid(mesh):mesh.queue_free()
  renderer.entries.erase(id)
 if renderer.gpu_pipelines.has(id):
  renderer.gpu_pipelines[id].root.queue_free();renderer.gpu_pipelines.erase(id)

func _show(id,actor: Dictionary,kind: String,state: int,attachments: Array):
 if renderer.entries.has(id) and renderer.entries[id].actor.get("root")!=actor.get("root"):_retire(id)
 var ok: bool=renderer.set_actor_state(id,actor,kind,state,attachments)
 if ok and renderer.entries.has(id) and renderer.entries[id].material is ShaderMaterial:
  var material: ShaderMaterial=renderer.entries[id].material
  if not material.has_meta("status_cull_compensated"):
   game.native_damage._compensate_reflected_culling(material);material.set_meta("status_cull_compensated",true)
 state_audit[id]={"state":state,"kind":kind,"source_state_bound":ok,"attachments":attachments.size(),"actor_binding":"displayed original minion" if kind=="minion" else "Original hero mesh registered to existing 2D hero position"}

 var go: Dictionary=hero_assets.objects[key]
 var tr: Dictionary=hero_assets.component(key,"Transform")

func sync_from_game(delta: float=0.0):
 if not is_instance_valid(renderer):return
 renderer.advance(delta)
 state_audit={}
 if game.game_over or game.help_visible or game.settings_visible or not game.flow.accepts_actions():clear("not_action_surface");return
 var aiming: bool=game._is_aiming() and game.player_turn and not game.busy
 var live: Dictionary={}
 for board in game._presentation_boards():
  for unit in board:
   var id: String="unit:%d"%int(unit.id);live[id]=true
   var state: int=1
   var visible: bool=not unit.get("dead",true) and not unit.get("hidden",false) and not unit.get("native_summon_hidden",false) and int(unit.hp)>0
   if visible:
    var hovered: bool=game.hover_unit==unit.id
    if aiming:
     # Original response mode3 does NOT fall through to ordinary options.
     if game._target_is_legal({"type":"unit","unit":unit,"pos":unit.pos}):state=27 if hovered else 26
     elif hovered:state=2
    elif int(unit.side)==0 and game.player_turn and game.can_attack(unit):state=7 if hovered else 6
    elif hovered:state=2
   if game.headless:state_audit[id]={"state":state,"kind":"minion","gpu":false};continue
   var piece: Dictionary=game.spatial_scene.pieces.get(int(unit.id),{})
   if piece.get("native_actor",false):_show(id,piece.actor,"minion",state,_attachments(id,piece.actor))
   else:renderer.hide(id)
 for side in [0,1]:
  var id: String="hero:%d"%side;live[id]=true
  var pos: Vector2=game.PLAYER_HERO if side==0 else game.ENEMY_HERO
  var target: Dictionary=game.target_at(game.pointer,side)
  var hovered: bool=target.get("type","")=="hero"
  var state: int=27 if hovered else 26
  if not aiming or not game._target_is_legal({"type":"hero","side":side,"pos":pos}):state=2 if hovered else 1
  if game.headless:state_audit[id]={"state":state,"kind":"hero","gpu":false};continue
  if state==1 and not heroes.has(side):continue
  var actor: Dictionary=_hero(side)
  heroes[side].holder.position=game.spatial_scene.screen_to_world(pos+game.shake_offset)
  _show(id,actor,"hero",state,_attachments(id,actor))
 for id in renderer.entries.keys():
  if not live.has(id):_retire(id)
 while renderer.history.size()>256:renderer.history.pop_front()

func clear(_reason: String="clear"):
 if is_instance_valid(renderer):renderer.hide_all()
 state_audit={}

Shared fixed-board source projection, no visual fit tuning.","rules_mutated":false,"hand_drawn_eligibility":false}
