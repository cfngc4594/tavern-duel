struct Mtl_VertexIn
{
    float4 POSITION0 [[ attribute(0) ]] ;
    float2 TEXCOORD0 [[ attribute(1) ]] ;
};

struct Mtl_VertexOut
{
    float4 mtl_Position [[ position, invariant ]];
    float2 TEXCOORD0 [[ user(TEXCOORD0) ]];
    float2 TEXCOORD1 [[ user(TEXCOORD1) ]];
    float2 TEXCOORD2 [[ user(TEXCOORD2) ]];
    float2 TEXCOORD3 [[ user(TEXCOORD3) ]];
    float2 TEXCOORD4 [[ user(TEXCOORD4) ]];
    float2 TEXCOORD5 [[ user(TEXCOORD5) ]];
    float2 TEXCOORD6 [[ user(TEXCOORD6) ]];
    float2 TEXCOORD7 [[ user(TEXCOORD7) ]];
};

vertex Mtl_VertexOut xlatMtlMain(
    constant VGlobals_Type& VGlobals [[ buffer(0) ]],
    constant UnityPerDraw_Type& UnityPerDraw [[ buffer(1) ]],
    constant UnityPerMaterial_Type& UnityPerMaterial [[ buffer(2) ]],
    Mtl_VertexIn input [[ stage_in ]])
{
    Mtl_VertexOut output;
    float4 u_xlat0;
    float4 u_xlat1;
    u_xlat0 = input.POSITION0.yyyy * UnityPerDraw.hlslcc_mtx4x4unity_ObjectToWorld[1];
    u_xlat0 = fma(UnityPerDraw.hlslcc_mtx4x4unity_ObjectToWorld[0], input.POSITION0.xxxx, u_xlat0);
    u_xlat0 = fma(UnityPerDraw.hlslcc_mtx4x4unity_ObjectToWorld[2], input.POSITION0.zzzz, u_xlat0);
    u_xlat0 = u_xlat0 + UnityPerDraw.hlslcc_mtx4x4unity_ObjectToWorld[3];
    u_xlat1 = u_xlat0.yyyy * VGlobals.hlslcc_mtx4x4unity_MatrixVP[1];
    u_xlat1 = fma(VGlobals.hlslcc_mtx4x4unity_MatrixVP[0], u_xlat0.xxxx, u_xlat1);
    u_xlat1 = fma(VGlobals.hlslcc_mtx4x4unity_MatrixVP[2], u_xlat0.zzzz, u_xlat1);
    output.mtl_Position = fma(VGlobals.hlslcc_mtx4x4unity_MatrixVP[3], u_xlat0.wwww, u_xlat1);
    u_xlat0.xy = input.TEXCOORD0.xy + float2(UnityPerMaterial._OffsetX, UnityPerMaterial._OffsetY);
    output.TEXCOORD0.xy = fma(u_xlat0.xy, UnityPerMaterial._MainTex_ST.xy, UnityPerMaterial._MainTex_ST.zw);
    u_xlat0.zw = input.TEXCOORD0.xy + (-float2(UnityPerMaterial._OffsetX, UnityPerMaterial._OffsetY));
    output.TEXCOORD1.xy = fma(u_xlat0.zw, UnityPerMaterial._MainTex_ST.xy, UnityPerMaterial._MainTex_ST.zw);
    output.TEXCOORD2.xy = fma(u_xlat0.xw, UnityPerMaterial._MainTex_ST.xy, UnityPerMaterial._MainTex_ST.zw);
    output.TEXCOORD3.xy = fma(u_xlat0.zy, UnityPerMaterial._MainTex_ST.xy, UnityPerMaterial._MainTex_ST.zw);
    u_xlat1.yw = u_xlat0.yw;
    u_xlat1.xz = input.TEXCOORD0.xx;
    output.TEXCOORD4.xy = fma(u_xlat1.xy, UnityPerMaterial._MainTex_ST.xy, UnityPerMaterial._MainTex_ST.zw);
    output.TEXCOORD5.xy = fma(u_xlat1.zw, UnityPerMaterial._MainTex_ST.xy, UnityPerMaterial._MainTex_ST.zw);
    u_xlat0.yw = input.TEXCOORD0.yy;
    output.TEXCOORD6.xy = fma(u_xlat0.xy, UnityPerMaterial._MainTex_ST.xy, UnityPerMaterial._MainTex_ST.zw);
    output.TEXCOORD7.xy = fma(u_xlat0.zw, UnityPerMaterial._MainTex_ST.xy, UnityPerMaterial._MainTex_ST.zw);
    return output;
}


#include <metal_stdlib>
#include <metal_texture>
using namespace metal;
constant uint32_t rp_output_remap_mask [[ function_constant(1) ]];
constant const uint rp_output_remap_0 = (rp_output_remap_mask >> 0) & 0xF;
struct FGlobals_Type
{
    float2 _GlobalMipBias;
};

struct UnityPerMaterial_Type
{
    float4 _MainTex_ST;
    float4 _Color;
    float _OffsetX;
    float _OffsetY;
};

struct Mtl_FragmentIn
{
    float2 TEXCOORD0 [[ user(TEXCOORD0) ]] ;
    float2 TEXCOORD1 [[ user(TEXCOORD1) ]] ;
    float2 TEXCOORD2 [[ user(TEXCOORD2) ]] ;
    float2 TEXCOORD3 [[ user(TEXCOORD3) ]] ;
    float2 TEXCOORD4 [[ user(TEXCOORD4) ]] ;
    float2 TEXCOORD5 [[ user(TEXCOORD5) ]] ;
    float2 TEXCOORD6 [[ user(TEXCOORD6) ]] ;
    float2 TEXCOORD7 [[ user(TEXCOORD7) ]] ;
};

struct Mtl_FragmentOut
{
    float4 SV_TARGET0 [[ color(rp_output_remap_0) ]];
};

fragment Mtl_FragmentOut xlatMtlMain(
    constant FGlobals_Type& FGlobals [[ buffer(0) ]],
    constant UnityPerMaterial_Type& UnityPerMaterial [[ buffer(1) ]],
    sampler sampler_MainTex [[ sampler (0) ]],
    texture2d<float, access::sample > _MainTex [[ texture(0) ]] ,
    Mtl_FragmentIn input [[ stage_in ]])
{
    Mtl_FragmentOut output;
    float u_xlat0;
    float4 u_xlat1;
    float u_xlat2;
    u_xlat0 = _MainTex.sample(sampler_MainTex, input.TEXCOORD0.xy, bias(FGlobals._GlobalMipBias.xyxx.x)).w;
    u_xlat2 = _MainTex.sample(sampler_MainTex, input.TEXCOORD1.xy, bias(FGlobals._GlobalMipBias.xyxx.x)).w;
    u_xlat0 = u_xlat2 + u_xlat0;
    u_xlat2 = _MainTex.sample(sampler_MainTex, input.TEXCOORD2.xy, bias(FGlobals._GlobalMipBias.xyxx.x)).w;
    u_xlat0 = u_xlat2 + u_xlat0;
    u_xlat2 = _MainTex.sample(sampler_MainTex, input.TEXCOORD3.xy, bias(FGlobals._GlobalMipBias.xyxx.x)).w;
    u_xlat0 = u_xlat2 + u_xlat0;
    u_xlat2 = _MainTex.sample(sampler_MainTex, input.TEXCOORD4.xy, bias(FGlobals._GlobalMipBias.xyxx.x)).w;
    u_xlat0 = u_xlat2 + u_xlat0;
    u_xlat2 = _MainTex.sample(sampler_MainTex, input.TEXCOORD5.xy, bias(FGlobals._GlobalMipBias.xyxx.x)).w;
    u_xlat0 = u_xlat2 + u_xlat0;
    u_xlat2 = _MainTex.sample(sampler_MainTex, input.TEXCOORD6.xy, bias(FGlobals._GlobalMipBias.xyxx.x)).w;
    u_xlat0 = u_xlat2 + u_xlat0;
    u_xlat2 = _MainTex.sample(sampler_MainTex, input.TEXCOORD7.xy, bias(FGlobals._GlobalMipBias.xyxx.x)).w;
    u_xlat0 = u_xlat2 + u_xlat0;
    u_xlat0 = u_xlat0 * 0.125;
    u_xlat1.xyz = float3(u_xlat0) * UnityPerMaterial._Color.xyz;
    u_xlat1.w = u_xlat0 * u_xlat0;
    output.SV_TARGET0 = u_xlat1 * UnityPerMaterial._Color.wwww;
    return output;
}
